## Practical 1: AngularJS Multiple Filters

index.html
<!DOCTYPE html>
<html>
<head>
 <title>AngularJS Filters Example</title>
 <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.6.9/angular.min.js"></script> 
 <link rel="stylesheet" href="style.css">
</head>

<body ng-app="filterApp" ng-controller="filterController">
 <div class="container">
 <h1>AngularJS Filters Example</h1>

 <h2>Search and Sort:</h2>
 <input type="text" ng-model="searchText" placeholder="Search for a fruit...">
 <select ng-model="sortColumn">
 <option value="">View options</option>
 <option value="name">Name</option>
 <option value="price">Price</option>
 </select>

 <ul>
 <li ng-repeat="fruit in fruits | filter:searchText | orderBy:sortColumn">
 {{ fruit.name | uppercase }} - {{ fruit.price | currency }} 
 </li>
 </ul>

 <h2>Date Filter:</h2>
 <p>{{ today | date:'fullDate' }}</p>

 <h2>JSON Filter:</h2>
 <pre>{{ product | json }}</pre>

 <h2>LimitTo Filter:</h2>
 <input type="number" ng-model="limitValue">
 <p>{{ message | limitTo:limitValue }}</p>

 <h2>Lowercase Filter:</h2>
 <input type="text" ng-model="lowercaseInput">
 <p>{{ lowercaseInput | lowercase }}</p>

 <h2>Uppercase Filter:</h2>
 <input type="text" ng-model="uppercaseInput">
 <p>{{ uppercaseInput | uppercase }}</p>

 <h2>Positive Number Filter: {{ product.price}}</h2>
 <p>{{ product.price | number:3 }}</p>

 <h2>Negative Number Filter: {{ product.price}}</h2>
 <p>{{ product.price/ 10 }}</p>

 </div>

 <script>
 var app = angular.module('filterApp', []); 

 app.controller('filterController', function($scope) { 
 $scope.product = { 
 name: 'Laptop', 
 price: 1234.56
 }; 

 $scope.today = new Date(); 

 $scope.fruits = [ 
 { name: 'Apple', price: 1.2 }, 
 { name: 'Banana', price: 0.5 }, 
 { name: 'Cherry', price: 3.0 }, 
 { name: 'Date', price: 2.5 }, 
 { name: 'Elderberry', price: 4.0 } 
 ]; 

 $scope.message = "Hello, AngularJS Filters!"; 

 $scope.limitValue = 10; 
 $scope.lowercaseInput = ''; 
 $scope.uppercaseInput = ''; 
 }); 
 </script>

</body>
</html>

## Practical 2: AngularJS Routing / SPA

Practical_2_Ajs_Routing.html

```html
<!DOCTYPE html>
<html>
<head>
 <title>AngularJs Routing</title>
 <script src="http://ajax.googleapis.com/ajax/libs/angularjs/1.4.8/angular.min.js"></script>
 <script src="http://ajax.googleapis.com/ajax/libs/angularjs/1.4.8/angular-route.min.js"></script>
</head>

<body>
<h2>AngularJS Routing / Single Page Application</h2>

<div ng-app="mainApp">
 <p><a href="#addStudent">Add Student</a></p>
 <p><a href="#viewStudents">View Students</a></p>
 <div ng-view></div>

 <script type="text/ng-template" id="addStudent.html">
 <h2>Add Student</h2>
 <p>{{message}}</p>
 </script>

 <script type="text/ng-template" id="viewStudents.html">
 <h2>View Students</h2>
 <p>{{message}}</p>
 </script>
</div>

<script>
var mainApp = angular.module("mainApp", ['ngRoute']);

mainApp.config(function($routeProvider){
 $routeProvider
 .when('/addStudent',{
 templateUrl:'addStudent.html',
 controller:'AddStudentController'
 })
 .when('/viewStudents',{
 templateUrl:'viewStudents.html',
 controller:'ViewStudentsController'
 })
 .otherwise({redirectTo:'/addStudent'});
});

mainApp.controller('AddStudentController',function($scope){
 $scope.message="Add Student Page";
});

mainApp.controller('ViewStudentsController',function($scope){
 $scope.message="View Students Page";
});
</script>

</body>
</html>
```

## Practical 4: Custom Validation

Custom_Validation.html

```html
<!DOCTYPE html>
<html>
<head>
 <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.6.9/angular.min.js"></script>
</head>

<body ng-app="myApp" ng-controller="myCtrl">

<form name="myForm" ng-submit="submitForm()" novalidate>

<input type="text" name="fullname" ng-model="formData.fullname" my-directive1 required>
<span ng-show="myForm.fullname.$invalid">Invalid</span>

<input type="submit">
</form>

<script>
var app = angular.module('myApp', []);

app.directive('myDirective1', function(){
 return {
 require:'ngModel',
 link:function(scope,element,attr,mCtrl){
 function validate(value){
 if(value.includes('a')){
 mCtrl.$setValidity('charE',true);
 }else{
 mCtrl.$setValidity('charE',false);
 }
 return value;
 }
 mCtrl.$parsers.push(validate);
 }
 };
});

app.controller('myCtrl', function($scope){
 $scope.submitForm=function(){
 if($scope.myForm.$valid){ alert("Success"); }
 };
});
</script>

</body>
</html>
```

## Practical 5: Parent Child React

App.js

```js
import React from 'react';
import Child from './Child';

class App extends React.Component {
 state={name:""}

 handleCallback=(data)=>{
 this.setState({name:data})
 }

 render(){
 return(
 <>
 <Child parentCallback={this.handleCallback}/>
 <p>{this.state.name}</p>
 </>
 )
 }
}

export default App;
```

Child.js

```js
import React from 'react';

class Child extends React.Component{
 onTrigger=(e)=>{
 this.props.parentCallback(e.target.myname.value);
 e.preventDefault();
 }

 render(){
 return(
 <form onSubmit={this.onTrigger}>
 <input name="myname"/>
 <input type="submit"/>
 </form>
 )
 }
}

export default Child;
```
